Here, we give generalizations of \cref{sync_max:thm:prim} 
to strong $\{a,b\}$-primitivity, but formulated
in a purely group-theoretical language without any 
reference to automata.
An automata-theoretical formulation 
in the spirit of \cref{sync_max:thm:prim} %(as it relates to \cref{sync_max:cor:prim_grp}, see also \cref{prelim:remarks_trans_semigrp_semiautomata})
can be easily given via the constructions from \cref{prelim:remarks_trans_semigrp_semiautomata}.
% according to the same scheme
%as in \THM~\ref{sync_max:thm:prim}). \todo{das in einführung sagen.}
Specifically, we characterize strong $\{a,b\}$-primitivity
by
\begin{enumerate}
	\item complete reachability and complete distinguishability,
	
	\item the property of having a transformation mapping
	distinct subsets to subsets of different cardinality,
	
	\item distinguishability of pairs of disjoint subsets of fixed cardinalities in case of a transitive permutation group,

    \item distinguishability of pairs of subsets of fixed cardinalities.
\end{enumerate}

So, we give further characterizations
than merely those suggested by \cref{sync_max:thm:prim}
for primitivity. From these, we also obtain, via~\cref{sync_max:prop:primitive_Aprimitive}, 
characterizations of primitivity; but as written in the introductory paragraph
to \cref{sec:A-primitive} we do not state them explicitly as they follow readily.

As in \cref{syncmax:rem:square-brackets}, here, we combine two statements into one using square
brackets. % (note that there exist only two simple idempotent mapping two points to a common point).

\begin{theorem}
	\label{sync_max:thm:stronglyab-idempotent}
	Let $G \le_G \mathcal S_n$ with $n \ge 5$ % erste drei auch für n \ge 3
	and $a,b \in [n]$ be distinct.
	Then the following are equivalent.
	\begin{enumerate}
		\item \label{sync_max:thm:stronglyab-idempotent:item:abstrong} $G$ is strongly $\{a,b\}$-primitive.	
		
		\item \label{sync_max:thm:stronglyab-idempotent:item:reach} For every [idempotent] $t \colon [n] \to [n]$ of defect one
		with $\applyfunc{a}{t} = \applyfunc{b}{t}$
		the transformation semigroup $\langle G \cup \{t\} \rangle_S$
		is completely reachable.
		
		\item \label{sync_max:thm:stronglyab-idempotent:item:dist} For every [idempotent] $t \colon [n] \to [n]$ of defect one
		with $\applyfunc{a}{t} = \applyfunc{b}{t}$
		the transformation semigroup $\langle G \cup \{t\} \rangle_S$
		is completely distinguishable.		
		
		\item \label{sync_max:thm:stronglyab-idempotent:item:card} For every [idempotent] $t \colon [n] \to [n]$ of defect one 
		with $t(a) = t(b)$ we have $|h(A)| \ne |h(B)|$ for all distinct non-singletons $A,B \subseteq [n]$
		for some $h \in \langle G \cup \{t\} \rangle_S$.
		
		\item \label{sync_max:thm:stronglyab-idempotent:item:trans-disj} The group $G$ is transitive and for some simple idempotent
		$t \colon [n] \to [n]$
		with $t(a) = t(b)$ every $k$-set is distinguishable from every disjoint $l$-set for some $2 \le k,l < n$ with $kl < n$
		by some element from $\langle G \cup \{t\} \rangle_S$.		
				
		\item \label{sync_max:thm:stronglyab-idempotent:item:kl} For every [idempotent] $t \colon [n] \to [n]$ of defect one with $t(a) = t(b)$ 
		every $k$-set is distinguishable from every distinct $l$-set for some $2 \le k,l < n$ by some element from $\langle G \cup \{t\} \rangle_S$.	
		
		\item \label{syncmax:stronglyab-reach:item:two-sets} For every [idempotent] $t \colon [n] \to [n]$ of defect one with $t(a) = t(b)$ 
	  	every $k$-set is reachable in $\langle G \cup \{t\} \rangle_S$
	  	for some $k \ge 2$ with $n > k(k-1)$.
	\end{enumerate} 
\end{theorem}
\begin{proof}
\ref{sync_max:thm:stronglyab-idempotent:item:abstrong} $\Rightarrow$ \ref{sync_max:thm:stronglyab-idempotent:item:reach}: First, assume $G$ is strongly $\{a,b\}$-primitive. Suppose $t \colon [n] \to [n]$ 
has defect one with $\applyfunc{a}{t} = \applyfunc{b}{t}$.
By Lemma~\ref{sync_words:lem:idempotent} we can assume~$t$ is a simple idempotent mapping (thus the same arguments works when~$t$
is assumed to be a simple idempotent).
Set $M = \langle G \cup \{t\}\rangle_S$, which is a monoid since~$G$ contains the identity transformation.
The mapping~$t$ induces the edge $(a,b)$ in the digraph~$I(M)$.
By Proposition~\ref{prelim:prop:group_of_units} the group $G$ is precisely the group of units of $M$.
As $G$ is a subgroup of the automorphism group of~$I(M)$ (Proposition~\ref{sync_max:prop:G_automorhism_on_graphs})
the edge set of $I(M)$ is a union of edge sets of orbital digraphs of $G$.
Hence, having $(a,b)$ as an edge in $I(M)$ implies that the orbital digraph of $(a,b)$ is a subgraph of $I(M)$.
As~$G$ is strongly $\{a,b\}$-primitive, 
by \cref{sync_max:prop:stronglyabprim_orbitalSCC}, the orbital digraph of $(a,b)$, and so $I(M)$, is strongly connected.
Then the claim follows from \cref{sync_max:thm:IS_SCC}.

\ref{sync_max:thm:stronglyab-idempotent:item:reach} $\Rightarrow$ \ref{sync_max:thm:stronglyab-idempotent:item:dist}: This follows from \cref{cor:single-perm} (reformulated for transformation semigroups, see \cref{prelim:remarks_trans_semigrp_semiautomata})
 and since sync-maximality implies complete distinguishability by \cref{sync_max:thm:char_sync_max_aut}.\footnote{This also
 	follows similarly as Item~\ref{sync_max:thm:stronglyab-idempotent:item:reach} from Item~\ref{sync_max:thm:stronglyab-idempotent:item:abstrong}.
 	With the notation from the proof of Item~\ref{sync_max:thm:stronglyab-idempotent:item:reach}, 
 	since $G$ is strongly $\{a,b\}$-primitive, by \cref{sync_max:prop:ab-prim-transitive}, it is 
 	transitive. So the orbital digraph of $(a,b)$ (which is strongly connected as shown in the proof of Item~\ref{sync_max:thm:stronglyab-idempotent:item:reach})
 	is a vertex-transitive digraph.
 	By Proposition~\ref{prelim:prop:vertex-transitive-SCC-32-connected} 
 	it is strongly $\frac{3}{2}$-connected.
 	So $I(M)$ is strongly~$\frac{3}{2}$-connected. 
 	Then Item~\ref{sync_max:thm:stronglyab-idempotent:item:dist} follows from Theorem~\ref{sync_max:thm:IS_32SCC}.} 	

\ref{sync_max:thm:stronglyab-idempotent:item:dist} $\Rightarrow$ \ref{sync_max:thm:stronglyab-idempotent:item:card}: This follows
 since an element distinguishing two subsets maps them to sets of distinct cardinalities.
	
\ref{sync_max:thm:stronglyab-idempotent:item:card} $\Rightarrow$ \ref{sync_max:thm:stronglyab-idempotent:item:trans-disj}:
 Let $A = \Orbit{a}{G} \cup \Orbit{b}{G}$.
 Assume there exists $c \in [n] \setminus A$.
 Then $\{a,c\}$ and $\{b,c\}$ are not distinguishable
 in $\langle G \cup \{t\} \rangle_S$ for any $t \colon [n] \to [n]$
 of defect one with $t(a) = t(b)$.
 Thus, we have $[n] = A$.
 Next, assume $\Orbit{a}{G} \ne \Orbit{b}{G}$
 and, without loss of generality, $|\Orbit{a}{G}| \ge 3$.
 Then for distinct $c,d \in \Orbit{a}{G}$ with $c \ne a$ and $d \ne a$,
 the sets $\{a,c\}$ and $\{a,d\}$ are not distinguishable
 in $\langle G \cup \{t\} \rangle_S$
 for the simple idempotent $t$ with $t(b) = a$.
 Thus, we must have $|\Orbit{a}{G}| \le 2$
 and $|\Orbit{b}{G}| \le 2$,
 which is not possible since $n \ge 5$ by assumption.
 Hence~$G$ must be transitive.
 
 Next, we choose some simple idempotent $t \colon [n] \to [n]$. Then, the assumption yields that pairs of distinct $2$-sets
 are distinguishable. So, the claim holds with $k = l = 2$.
 
\ref{sync_max:thm:stronglyab-idempotent:item:trans-disj} $\Rightarrow$ \ref{sync_max:thm:stronglyab-idempotent:item:abstrong}: 
Let $t \colon [n] \to [n]$ be some simple idempotent
with $t(a) = t(b)$ such that in $\langle G \cup \{t\} \rangle_S$
all pairs of disjoint $k$- and $l$-sets are distinguishable
for $2 \le k,l < n$ with $kl < n$. 
Then, by~\cref{sync_max:prop:disjointklsets}
$\langle G \cup \{t\} \rangle_S$ contains a constant map.
\cref{sync_max:cor:syncSI-abprim2}
yields that $G$ is $\{a,b\}$-primitive. So, as $G$ is transitive,
it is strongly $\{a,b\}$-primitive.	

\ref{sync_max:thm:stronglyab-idempotent:item:kl} $\Leftrightarrow$ \ref{sync_max:thm:stronglyab-idempotent:item:abstrong}:  
First, if $G$ is strongly $\{a,b\}$-primitive, then, as shown in Item~\ref{sync_max:thm:stronglyab-idempotent:item:dist},
$\langle G \cup \{t\} \rangle_S$ is completely distinguishable. 
 In particular,
every $k$-set is distinguishable from every distinct $l$-set
for all $k,l \ge 2$.
This, we can choose, for example, $k = l = 2$.

Conversely, without loss of generality, suppose $l \le k$.
For $l = k = 2$,
as $n \ge 5$,
we get strong $\{a,b\}$-primitivity from~\cref{sync_max:cor:2-sets_transitive}.
For $l = k$ with $k \ge 3$,
as $n \ge k + 1$, from \cref{sync_max:cor:k-ge-3-sync}
we get that $G$ synchronizes $t$.
For $l < k$, from 
\crefitem{sync_max:cor:two-sets-containing-2set}{sync_max:cor:two-sets-containing-2set:item:kl}
we get that $G$ synchronizes $t$.
In the latter two cases,
by \cref{sync_max:thm:idemp}, $G$ is strongly $\{a,b\}$-primitive.

Both implications hold whether we consider only simple idempotents~$t$
or any transformation~$t$ of defect one mapping $a$ and~$b$
to a common point.

\ref{syncmax:stronglyab-reach:item:two-sets} $\Leftrightarrow$ \ref{sync_max:thm:stronglyab-idempotent:item:abstrong}:
This follows from combining
\cref{sync_max:prop:reachability-k-sets-sync},
Item~\ref{sync_max:thm:stronglyab-idempotent:item:dist},
and \cref{syncmax:cor:Aprim-idempt}.
\end{proof}

\begin{corollary}
	A strongly $\{a,b\}$-primitive permutation group on $[n]$
	has at most $n - 2$ orbits on the $2$-sets.	
\end{corollary}
\begin{proof}
	By adding a simple idempotent,
	we obtain that the generated transformation semigroup
	is completely reachable by \cref{sync_max:thm:stronglyab-idempotent}.
	Then, by \cref{sync_max:subsets:prop:num-two-set-orbits},
	the number of orbits on the $2$-sets is bounded by $n - 2$.
\end{proof} 

\begin{remark}
	The first three equivalences in \cref{sync_max:thm:stronglyab-idempotent}
	also hold for $n \ge 3$.
\end{remark}

\begin{remark}
	Combining \cref{sync_aut:permgrp:ex:cyclicAprim}
	and \cref{sync_max:thm:stronglyab-idempotent}
	immediately gives an alternative proof of \cref{sync_max:thm:circular_sync_max}
	for $k = 1$.
\end{remark}

\begin{remark}
	Note that in \crefitem{sync_max:thm:stronglyab-idempotent}{sync_max:thm:stronglyab-idempotent:item:trans-disj}
	we only demand existence of some simple idempotent~$t$,
	and not that the written claim holds for all simple idempotents~$t$ (or mappings of defect one).
	However, in this case, the assumption of transitive cannot be omitted
    to obtain a characterization of $\{a,b\}$-primitivity.
	For example, consider $G = \langle (0 ~ 1 ~ \cdots ~ n - 1)(n)(n+1) \rangle_G$
	and the simple idempotent $t \colon [n+2] \to [n+2]$
	that maps $n-1$ to $n$. 
	Then in $\langle G \cup \{t\} \rangle_S$
	all disjoint pairs of $2$-sets are distinguishable,
	but $G$ is not $\{n-1,n\}$-primitive as $\{0,1,\ldots,n-1,n\}$ is a non-trivial block containing $\{n-1,n\}$.
\end{remark}

\begin{remark}
	Note that for transitive $G \le_G \mathcal S_n$,
	by \cref{sync_words:lem:idempotent},
	for both simple idempotents $t,h \colon [n] \to [n]$
	we have $\langle G \cup \{t\} \rangle_S = \langle G \cup \{h\} \rangle_S$.
	Since all conditions from \cref{sync_max:thm:stronglyab-idempotent} 
	imply transitivity of~$G$ for $n \ge 5$ and it is sufficient to consider simple idempotents,
	all of them are actually equivalent to $G$ being
	transitive and the condition only holding for a single
	simple idempotent.	
\end{remark}

\begin{remark}
	\Crefitem{sync_max:thm:stronglyab-idempotent}{sync_max:thm:stronglyab-idempotent:item:kl}
	is also equivalent to the same condition but where~$t$
	is only some transformation~$t$ of defect one with $t(a) = t(b)$.
	This follows since by \cref{sync_words:lem:idempotent},
	$\langle G \cup \{t\} \rangle_S$ contains both simple idempotents
	mapping $a$ and~$b$ to a common point
	and so the claim from \crefitem{sync_max:thm:stronglyab-idempotent}{sync_max:thm:stronglyab-idempotent:item:kl}
	if this is true and~$t$ only has defect one.		
\end{remark}

\begin{remark}
	The conditions from \cref{sync_max:thm:stronglyab-idempotent} cannot be weakened to give equivalences
	for $\{a,b\}$-primitivity only, for example, by omitting
	transitivity in \crefitem{sync_max:thm:stronglyab-idempotent}{sync_max:thm:stronglyab-idempotent:item:kl}.
	For instance, the transformation monoids $\TS{\mathcal A}$ of the automata $\mathcal A = (Q, \Sigma, \delta)$ 
	from \cref{prelim:fig:binarySI-sync-trap}
    are $\{s,t\}$-primitive	since if $s \sim t$, then $\delta(s, b^i) \sim \delta(t, b^i) = t$ for all $i \ge 0$
	and $\{ \delta(s, b^i) \mid i \ge 0 \} = Q \setminus \{t\}$, which yields
	that all states are congruent to~$t$.
	However, in this automata pairs of sets of the form $A$ and $A \cup \{t\}$
	for $|A| \ge 2$ are indistinguishable.
\end{remark}
