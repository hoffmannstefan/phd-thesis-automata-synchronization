\def\@part[#1]#2{%
	\ifnum \c@secnumdepth >-2\relax
	\refstepcounter{part}%
	\addcontentsline{toc}{part}{\partname\
		\thepart\thechapterend\hspace{\betweenumberspace}#1}%
	\else
	\addcontentsline{toc}{part}{#1}%
	\fi
	\markboth{}{}%
	{\raggedleft
		\hyphenpenalty \@M
		\interlinepenalty\@M
		\ifnum \c@secnumdepth >-2\relax
		\normalfont\partnumsize\partnumstyle
		\partname\ \thepart
		\par
		\vskip 2\p@
		\fi
		\partsize\partstyle #2\par
		\partquote
	}%
	\global\let\partquote\@empty
	\@endpart
}
\makeatother

\newcommand{\partwithquote}[3]{%
	\setpartquote{#2}{#3}%
	\part{#1}%
}
